uint8_t dec2bcd(uint8_t val) { return ((val / 10) << 4) + (val % 10); }
uint8_t bcd2dec(uint8_t val) { return ((val >> 4) * 10) + (val & 0x0F); }

static void time_sync_notification_cb(struct timeval *tv) {
    ESP_LOGI(TAG, "SNTP Sincronizado com a internet! Gravando no RTC...");
    time_t now;
    struct tm timeinfo;
    time(&now);
    localtime_r(&now, &timeinfo);
    
    if (rtc_handle) {
        uint8_t data[8];
        data[0] = 0x04; 
        data[1] = dec2bcd(timeinfo.tm_sec);
        data[2] = dec2bcd(timeinfo.tm_min);
        data[3] = dec2bcd(timeinfo.tm_hour);
        data[4] = dec2bcd(timeinfo.tm_mday);
        data[5] = dec2bcd(timeinfo.tm_wday);
        data[6] = dec2bcd(timeinfo.tm_mon + 1);    
        data[7] = dec2bcd(timeinfo.tm_year % 100); 
        
        i2c_master_transmit(rtc_handle, data, 8, 1000);
        ESP_LOGI(TAG, "RTC atualizado com sucesso: %02d:%02d:%02d", timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec);
    }
}

static void initialize_sntp(void) {
    ESP_LOGI(TAG, "Iniciando servico NTP...");
    esp_sntp_setoperatingmode(SNTP_OPMODE_POLL);
    esp_sntp_setservername(0, "pool.ntp.org");
    sntp_set_time_sync_notification_cb(time_sync_notification_cb);
    esp_sntp_init();
    
    setenv("TZ", "<-03>3", 1);
    tzset();
}

static void set_wifi_state(bool state) {
    wifi_started = state;
    if (state) {
        ESP_LOGI(TAG, "Wi-Fi: Ligando antena...");
        esp_wifi_start();
        if (btn_wifi_settings) lv_obj_set_style_bg_color(btn_wifi_settings, lv_color_hex(0x007BFF), 0);
        if (sw_wifi_list) lv_obj_add_state(sw_wifi_list, LV_STATE_CHECKED);
        if (icon_wifi_watchface) lv_obj_set_style_text_color(icon_wifi_watchface, lv_color_white(), 0); 
        if (icon_wifi_slash) lv_obj_add_flag(icon_wifi_slash, LV_OBJ_FLAG_HIDDEN); // Esconde a barra
    } else {
        ESP_LOGI(TAG, "Wi-Fi: Desligando antena...");
        esp_wifi_stop();
        if (btn_wifi_settings) lv_obj_set_style_bg_color(btn_wifi_settings, lv_color_hex(0x333333), 0);
        if (sw_wifi_list) lv_obj_remove_state(sw_wifi_list, LV_STATE_CHECKED);
        if (icon_wifi_watchface) lv_obj_set_style_text_color(icon_wifi_watchface, lv_color_hex(0x333333), 0); 
        if (icon_wifi_slash) lv_obj_remove_flag(icon_wifi_slash, LV_OBJ_FLAG_HIDDEN); // Mostra a barra
    }
}

// Timer para dar tempo da tela renderizar antes do gargalo de hardware
static void delayed_usb_start_cb(lv_timer_t * t) {
    // Passa o controle para o PC apenas agora
    SdUsbManager::get_instance().enable_usb_msc();
    lv_timer_delete(t); // Timer de disparo Ãºnico
}

// FunÃ§Ã£o para alternar o modo Pendrive
static void set_usb_msc_state(bool state) {
    if (state == usb_msc_started) return; // Evita loop de estado

    usb_msc_started = state;
    
    if (state) {
        ESP_LOGI(TAG, "USB MSC: Solicitando ativaÃ§Ã£o com delay visual...");
        if (btn_usb_settings) lv_obj_set_style_bg_color(btn_usb_settings, lv_color_hex(0x007BFF), 0);
        
        // Vai para a tela estÃ¡tica SEM animaÃ§Ã£o para evitar processamento grÃ¡fico (DMA) simultÃ¢neo ao USB
        lv_scr_load_anim(scr_usb_msc, LV_SCR_LOAD_ANIM_NONE, 0, 0, false);
        
        // DÃ¡ 300ms para o LVGL limpar o buffer e desenhar a tela antes do USB assumir a CPU
        lv_timer_create(delayed_usb_start_cb, 300, NULL);
    } else {
        ESP_LOGI(TAG, "USB MSC: Desligando Modo Pendrive...");
        
        // Retoma o controle para o RelÃ³gio (Vai causar o Restart para recuperar o Serial)
        SdUsbManager::get_instance().disable_usb_msc();
    }
}

static void home_btn_timer_cb(lv_timer_t *timer) {
    static uint8_t last_state = 1;
    uint8_t current_state = gpio_get_level(BOOT_BTN_PIN);
    
    // 1. VerificaÃ§Ã£o do BotÃ£o FÃ­sico
    if (last_state == 1 && current_state == 0) { 
        if (lv_screen_active() != scr_watchface) {
            lv_scr_load_anim(scr_watchface, LV_SCR_LOAD_ANIM_FADE_ON, 200, 0, false);
        }
        // Acorda a tela caso o usuÃ¡rio aperte o botÃ£o no escuro
        reset_inactivity_timer(); 
    }
    last_state = current_state;

    // 2. [NOVO] VerificaÃ§Ã£o Global de Toque na Tela
    // Pega o ponteiro do driver de touch ativo no LVGL
    lv_indev_t * indev = lv_indev_get_next(NULL); 
    if (indev) {
        // Se a tela estiver sendo tocada em QUALQUER lugar, reseta o timer
        if (lv_indev_get_state(indev) == LV_INDEV_STATE_PRESSED) {
            reset_inactivity_timer();
        }
    }
}

static void update_time_timer_cb(lv_timer_t *timer) {
    if (!rtc_handle) return;
    uint8_t reg_addr = 0x04; 
    uint8_t data[7] = {0};   
    if (bsp_display_lock(pdMS_TO_TICKS(50))) {
        if (i2c_master_transmit_receive(rtc_handle, &reg_addr, 1, data, 7, 1000) == ESP_OK) {
            uint8_t sec = bcd2dec(data[0] & 0x7F);
            uint8_t min = bcd2dec(data[1] & 0x7F);
            uint8_t hour = bcd2dec(data[2] & 0x3F);
            uint8_t day = bcd2dec(data[3] & 0x3F);
            uint8_t month = bcd2dec(data[5] & 0x1F);
            uint16_t year = 2000 + bcd2dec(data[6]);
            if (time_label) lv_label_set_text_fmt(time_label, "%02d:%02d:%02d", hour, min, sec);
            if (date_label) lv_label_set_text_fmt(date_label, "%02d/%02d/%04d", day, month, year);
        }
        bsp_display_unlock();
    }
}

static void wifi_list_item_click_cb(lv_event_t * e) {
    lv_obj_t * btn = (lv_obj_t *)lv_event_get_target(e);
    lv_obj_t * label = lv_obj_get_child(btn, 1); 
    if (label) {
        const char * text = lv_label_get_text(label);
        const char * paren = strrchr(text, '(');
        if (paren != NULL && paren > text) {
            int len = paren - text - 1; 
            if (len > 32) len = 32;
            strncpy(current_ssid, text, len);
            current_ssid[len] = '\0';
        } else {
            strncpy(current_ssid, text, 32);
            current_ssid[32] = '\0';
        }

        ESP_LOGI(TAG, "Rede selecionada: %s", current_ssid);
        
        // Verifica se a rede jÃ¡ estÃ¡ salva no NVS
        wifi_config_t saved_config;
        esp_wifi_get_config(WIFI_IF_STA, &saved_config);

        char title_buf[64];
        if (strcmp((char*)saved_config.sta.ssid, current_ssid) == 0 && strlen((char*)saved_config.sta.password) > 0) {
            ESP_LOGI(TAG, "Rede ja salva. Preenchendo teclado e auto-conectando.");
            snprintf(title_buf, sizeof(title_buf), "Conectando: %s", current_ssid);
            lv_label_set_text(label_password_title, title_buf);
            
            // Preenche o teclado com a senha antiga (Ãºtil se a pessoa sÃ³ quiser ver ou editar)
            lv_textarea_set_text(ta_password, (char*)saved_config.sta.password);
            
            // Tenta conectar logo de cara
            esp_wifi_connect();
        } else {
            snprintf(title_buf, sizeof(title_buf), "Senha: %s", current_ssid);
            lv_label_set_text(label_password_title, title_buf);
            lv_textarea_set_text(ta_password, ""); 
        }

        lv_scr_load_anim(scr_password, LV_SCR_LOAD_ANIM_MOVE_LEFT, 300, 0, false);
    }
}

static void wifi_event_handler(void* arg, esp_event_base_t event_base, int32_t event_id, void* event_data) 
{
    // [NOVO] Auto-conectar ao ligar a antena
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
        ESP_LOGI(TAG, "Wi-Fi Iniciado. Tentando auto-conectar na rede salva...");
        esp_wifi_connect();
    }
    
    else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_SCAN_DONE) {
        uint16_t ap_count = 0;
        esp_wifi_scan_get_ap_num(&ap_count);
        
        if (ap_count == 0) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_obj_clean(list_wifi); 
                lv_obj_t * txt = lv_list_add_text(list_wifi, "Nenhuma rede encontrada");
                style_list_item(txt);
                bsp_display_unlock();
            }
            return;
        }

        if (ap_count > 15) ap_count = 15;
        wifi_ap_record_t *ap_info = (wifi_ap_record_t *)malloc(sizeof(wifi_ap_record_t) * ap_count);
        if (ap_info == NULL) return;

        if (esp_wifi_scan_get_ap_records(&ap_count, ap_info) == ESP_OK) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_obj_clean(list_wifi); 
                for (int i = 0; i < ap_count; i++) {
                    char ssid_str[33];
                    strncpy(ssid_str, (char *)ap_info[i].ssid, 32);
                    ssid_str[32] = '\0'; // <-- CORREÃ‡ÃƒO AQUI
                    if (strlen(ssid_str) == 0) continue;

                    char list_item_text[64];
                    snprintf(list_item_text, sizeof(list_item_text), "%s (%d dBm)", ssid_str, ap_info[i].rssi);

                    lv_obj_t * btn = lv_list_add_button(list_wifi, LV_SYMBOL_WIFI, list_item_text);
                    style_list_item(btn); 
                    lv_obj_add_event_cb(btn, wifi_list_item_click_cb, LV_EVENT_CLICKED, NULL);
                }
                bsp_display_unlock();
            }
        }
        free(ap_info);
    }
    
    else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "Desconectado do Wi-Fi.");
        
        if (wifi_started && icon_wifi_watchface) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_obj_set_style_text_color(icon_wifi_watchface, lv_color_white(), 0);
                bsp_display_unlock();
            }
        }

        if (lv_screen_active() == scr_password) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_label_set_text(label_password_title, "Falha! Senha Incorreta?");
                bsp_display_unlock();
            }
        }
    }
    
    else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t* event = (ip_event_got_ip_t*) event_data;
        ESP_LOGI(TAG, "Conectado! IP: " IPSTR, IP2STR(&event->ip_info.ip));
        
        // LÃª do chip de Wi-Fi o nome da rede que conectamos e salva na global
        wifi_ap_record_t ap_info;
        if (esp_wifi_sta_get_ap_info(&ap_info) == ESP_OK) {
            strncpy(current_ssid, (char *)ap_info.ssid, sizeof(current_ssid) - 1);
            current_ssid[sizeof(current_ssid) - 1] = '\0';
        }

        if (icon_wifi_watchface) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_obj_set_style_text_color(icon_wifi_watchface, lv_color_hex(0x00FF00), 0);
                bsp_display_unlock();
            }
        }

        static bool sntp_inited = false;
        if (!sntp_inited) {
            initialize_sntp();
            sntp_inited = true;
        }
        
        if (lv_screen_active() == scr_password) {
            if (bsp_display_lock(pdMS_TO_TICKS(100))) {
                lv_label_set_text(label_password_title, "Conectado!");
                lv_scr_load_anim(scr_settings, LV_SCR_LOAD_ANIM_MOVE_RIGHT, 400, 1000, false);
                bsp_display_unlock();
            }
        }
    }
}

static void start_wifi_scan(void) {
    if (bsp_display_lock(pdMS_TO_TICKS(100))) {
        lv_obj_clean(list_wifi); 
        lv_obj_t * txt = lv_list_add_text(list_wifi, "Escaneando...");
        style_list_item(txt);
        bsp_display_unlock();
    }
    wifi_scan_config_t scan_config = {};
    scan_config.show_hidden = false;
    scan_config.scan_type = WIFI_SCAN_TYPE_ACTIVE;
    scan_config.scan_time.active.min = 100;
    scan_config.scan_time.active.max = 300;
    esp_wifi_scan_start(&scan_config, false);
}

static void init_wifi_basic(void) {
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
      ESP_ERROR_CHECK(nvs_flash_erase());
      ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_sta();
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));
    
    esp_wifi_set_storage(WIFI_STORAGE_FLASH); 
    
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));

    ESP_ERROR_CHECK(esp_event_handler_instance_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, NULL));
    ESP_ERROR_CHECK(esp_event_handler_instance_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &wifi_event_handler, NULL, NULL));
}

static uint8_t read_battery_percentage(void) {
    // Apenas use o i2c_bus global que inicializamos na app_main
    if (!i2c_bus) return 0;
    
    i2c_device_config_t pmu_cfg = {};
    pmu_cfg.dev_addr_length = I2C_ADDR_BIT_LEN_7;
    pmu_cfg.device_address = AXP2101_ADDR;
    pmu_cfg.scl_speed_hz = 400000;
    
    i2c_master_dev_handle_t pmu_handle;
    if (i2c_master_bus_add_device(i2c_bus, &pmu_cfg, &pmu_handle) == ESP_OK) {
        uint8_t reg_addr = 0xA4;
        uint8_t batt_percent = 0;
        i2c_master_transmit_receive(pmu_handle, &reg_addr, 1, &batt_percent, 1, 1000);
        i2c_master_bus_rm_device(pmu_handle);
        return (batt_percent > 100) ? 100 : batt_percent;
    }
    return 0;
}

// Reseta o contador sempre que houver toque ou gesto nas telas
void reset_inactivity_timer(void) {
    last_activity_time = lv_tick_get();
    if (is_dimmed) {
        bsp_display_brightness_set(80); // Restaura o brilho padrao (80%)
        is_dimmed = false;
        ESP_LOGI(TAG, "Atividade detectada! Brilho restaurado.");
    }
}

// Deve ser associada as telas para monitorar interacoes fisicas
static void global_touch_intercept_cb(lv_event_t * e) {
    reset_inactivity_timer();
}

static void check_power_management_timer_cb(lv_timer_t * timer) {
    uint32_t elapsed = lv_tick_elaps(last_activity_time);
    
    // ObtÃ©m a tela atualmente ativa
    lv_obj_t * current_screen = lv_screen_active();
    
    // Define se estamos no "modo principal" ou no "modo pendrive"
    bool is_usb_screen = (current_screen == scr_usb_msc);
    bool is_main_interface = (current_screen == scr_watchface || 
                              current_screen == scr_settings || 
                              current_screen == scr_apps ||
                              current_screen == scr_wifi_list ||
                              current_screen == scr_password);

    // LÃ³gica do Brilho
    uint32_t dim_threshold = 10000; // PadrÃ£o 10s para Watchface/Main/USB
    if (!is_main_interface && !is_usb_screen) {
        dim_threshold = 30000; // Para Apps (Menu), 30s
    }

    // LÃ³gica de Desligamento Profundo (Deep Sleep)
    uint32_t off_threshold = 180000; // PadrÃ£o 3 minutos para todas as telas (180000ms)
    if (is_usb_screen) {
        // Se estiver no Pendrive, segura o relÃ³gio ligado por 4 horas (4 * 60 * 60 * 1000 = 14400000ms)
        off_threshold = 14400000; 
    }

    // LÃ³gica para diminuir o brilho
    if (elapsed >= dim_threshold && !is_dimmed) {
        bsp_display_brightness_set(10); // Reduz para 10%
        is_dimmed = true;
        ESP_LOGI(TAG, "Inatividade alcanÃ§ada. Esmaecendo tela...");
    }

    // LÃ³gica para desligar o relÃ³gio
    if (elapsed >= off_threshold) {
        ESP_LOGW(TAG, "Inatividade (Limite: %lum) alcanÃ§ada. Desligando sistema via AXP2101...", off_threshold / 60000);
        bsp_display_backlight_off();
        if (wifi_started) { esp_wifi_stop(); }
        
        if (i2c_bus) {
            i2c_device_config_t pmu_cfg = {};
            pmu_cfg.dev_addr_length = I2C_ADDR_BIT_LEN_7;
            pmu_cfg.device_address = 0x34; // AXP2101_ADDR
            pmu_cfg.scl_speed_hz = 400000;
            
            i2c_master_dev_handle_t pmu_handle;
            if (i2c_master_bus_add_device(i2c_bus, &pmu_cfg, &pmu_handle) == ESP_OK) {
                uint8_t reg_addr = 0x10;
                uint8_t val = 0;
                
                if (i2c_master_transmit_receive(pmu_handle, &reg_addr, 1, &val, 1, 1000) == ESP_OK) {
                    val |= 0x01; 
                    uint8_t data_write[2] = {0x10, val};
                    
                    ESP_LOGW(TAG, "Comando enviado! Cortando energia...");
                    i2c_master_transmit(pmu_handle, data_write, 2, 1000);
                } else {
                    ESP_LOGE(TAG, "Falha ao ler o AXP2101");
                }
                i2c_master_bus_rm_device(pmu_handle);
            }
        } else {
             ESP_LOGE(TAG, "I2C Bus estÃ¡ NULO! NÃ£o Ã© possÃ­vel desligar o PMU.");
        }
    }
    
    // Atualiza o valor da bateria dinamicamente na watchface a cada 1 segundo
    if (label_battery_watchface && current_screen == scr_watchface) {
        current_battery_pct = read_battery_percentage();
        
        // Define o sÃ­mbolo correto com base na porcentagem usando os sÃ­mbolos nativos do LVGL
        const char * sym = LV_SYMBOL_BATTERY_FULL;
        if (current_battery_pct <= 10) sym = LV_SYMBOL_BATTERY_EMPTY;
        else if (current_battery_pct <= 30) sym = LV_SYMBOL_BATTERY_1; // Equivale a 1/4
        else if (current_battery_pct <= 60) sym = LV_SYMBOL_BATTERY_2; // Equivale a 2/4 (Metade)
        else if (current_battery_pct <= 85) sym = LV_SYMBOL_BATTERY_3; // Equivale a 3/4

        lv_label_set_text_fmt(label_battery_watchface, "%s %d%%", sym, current_battery_pct);
        
        // Muda para vermelho se estiver acabando
        if (current_battery_pct <= 30) {
            lv_obj_set_style_text_color(label_battery_watchface, lv_color_hex(0xFF0000), 0);
        } else {
            lv_obj_set_style_text_color(label_battery_watchface, lv_color_white(), 0);
        }

        // DetecÃ§Ã£o de USB/Carregamento via I2C (AXP2101 Reg 0x00)
        bool usb_connected = false;
        if (i2c_bus) {
            i2c_device_config_t pmu_cfg = {};
            pmu_cfg.dev_addr_length = I2C_ADDR_BIT_LEN_7;
            pmu_cfg.device_address = 0x34;
            pmu_cfg.scl_speed_hz = 400000;
            
            i2c_master_dev_handle_t pmu_handle;
            if (i2c_master_bus_add_device(i2c_bus, &pmu_cfg, &pmu_handle) == ESP_OK) {
                uint8_t reg_status1 = 0x00;
                uint8_t val = 0;
                if (i2c_master_transmit_receive(pmu_handle, &reg_status1, 1, &val, 1, 1000) == ESP_OK) {
                    // Bit 5 indica presenÃ§a de VBUS (Cabo USB conectado)
                    usb_connected = (val & 0x20) != 0; 
                }
                i2c_master_bus_rm_device(pmu_handle);
            }
        }

        // Exibe ou esconde o raio amarelo baseado no status do cabo
        if (icon_charge_watchface) {
            if (usb_connected) {
                lv_obj_remove_flag(icon_charge_watchface, LV_OBJ_FLAG_HIDDEN);
                
                // Libera o botÃ£o (O LVGL muda a cor sozinho!)
                if (btn_usb_settings && lv_obj_has_state(btn_usb_settings, LV_STATE_DISABLED)) {
                    lv_obj_remove_state(btn_usb_settings, LV_STATE_DISABLED);
                }
            } else {
                lv_obj_add_flag(icon_charge_watchface, LV_OBJ_FLAG_HIDDEN);
                
                // Bloqueia o botÃ£o (O LVGL muda a cor sozinho!)
                if (btn_usb_settings && !lv_obj_has_state(btn_usb_settings, LV_STATE_DISABLED)) {
                    lv_obj_add_state(btn_usb_settings, LV_STATE_DISABLED);
                    if (usb_msc_started) set_usb_msc_state(false); // Garante o desligamento do pendrive
                }
            }
        }
    }
}

static void boot_fade_anim_cb(void * var, int32_t v) {
    lv_color_t bg = lv_color_mix(lv_color_black(), lv_color_white(), v);
    lv_color_t text_main = lv_color_mix(lv_color_white(), lv_color_black(), v);
    lv_color_t text_sec = lv_color_mix(lv_color_hex(0xAAAAAA), lv_color_black(), v);
    lv_color_t text_gray = lv_color_mix(lv_color_hex(0x333333), lv_color_black(), v);

    lv_obj_set_style_bg_color(scr_watchface, bg, 0);
    lv_obj_set_style_text_color(time_label, text_main, 0);
    lv_obj_set_style_text_color(date_label, text_sec, 0);
    
    // Wi-Fi: Se desligado termina cinza, se ligado termina branco
    if (!wifi_started) {
        lv_obj_set_style_text_color(icon_wifi_watchface, text_gray, 0);
        if (icon_wifi_slash) lv_obj_set_style_text_color(icon_wifi_slash, text_gray, 0);
    } else {
        lv_obj_set_style_text_color(icon_wifi_watchface, text_main, 0);
    }

    // Bateria: Se <= 30% termina vermelho, senÃ£o termina branco
    lv_color_t bat_target = (current_battery_pct <= 30) ? lv_color_hex(0xFF0000) : lv_color_white();
    lv_color_t bat_color = lv_color_mix(bat_target, lv_color_black(), v);
    lv_obj_set_style_text_color(label_battery_watchface, bat_color, 0);
    lv_color_t yellow_target = lv_color_hex(0xFFD700); // Amarelo Ouro
    lv_color_t yellow_color = lv_color_mix(yellow_target, lv_color_black(), v);
    if (icon_charge_watchface) {
        lv_obj_set_style_text_color(icon_charge_watchface, yellow_color, 0);
    }
}

// Disparador seguro que vai rodar na memÃ³ria do motor grÃ¡fico
static void start_anim_timer_cb(lv_timer_t * timer) {
    // Declaramos como static para nÃ£o pesar no stack
    static lv_anim_t a;
    lv_anim_init(&a);
    lv_anim_set_var(&a, scr_watchface);     
    lv_anim_set_values(&a, 0, 255);         
    lv_anim_set_time(&a, 1500);             
    lv_anim_set_exec_cb(&a, boot_fade_anim_cb);
    lv_anim_set_path_cb(&a, lv_anim_path_ease_out); 
    lv_anim_start(&a);
    
    // Deleta este timer logo apÃ³s ligar a animaÃ§Ã£o (tiro Ãºnico)
    lv_timer_delete(timer); 
}

#include "driver/gpio.h" // [NOVO] NecessÃ¡rio para manipular os pinos USB

// Pinos do SD Card confirmados via diagrama da Waveshare
#define PIN_NUM_MOSI 1
#define PIN_NUM_CLK  2
#define PIN_NUM_MISO 3
#define PIN_NUM_CS   17 

static const char *TAG = "SdUsbMgr";

esp_err_t SdUsbManager::init_local_storage() {
    if (sd_initialized && !msc_active) return ESP_OK;

    ESP_LOGI(TAG, "Montando Cartao SD no ESP32...");

    sdmmc_host_t host = SDSPI_HOST_DEFAULT();
    
    // ForÃ§a o uso do SPI3_HOST para nÃ£o conflitar com o Display AMOLED (SPI2_HOST)
    host.slot = SPI3_HOST; 
    
    spi_bus_config_t bus_cfg = {};
    bus_cfg.mosi_io_num = PIN_NUM_MOSI;
    bus_cfg.miso_io_num = PIN_NUM_MISO;
    bus_cfg.sclk_io_num = PIN_NUM_CLK;
    bus_cfg.quadwp_io_num = -1;
    bus_cfg.quadhd_io_num = -1;
    bus_cfg.max_transfer_sz = 4000;

    // Inicializa no SPI3. Ignora se jÃ¡ estiver ligado.
    esp_err_t err = spi_bus_initialize((spi_host_device_t)host.slot, &bus_cfg, SDSPI_DEFAULT_DMA);
    if (err != ESP_OK && err != ESP_ERR_INVALID_STATE) {
        ESP_LOGE(TAG, "Falha ao inicializar barramento SPI: %s", esp_err_to_name(err));
        return err;
    }

    sdspi_device_config_t slot_config = SDSPI_DEVICE_CONFIG_DEFAULT();
    slot_config.gpio_cs = (gpio_num_t)PIN_NUM_CS;
    slot_config.host_id = (spi_host_device_t)host.slot;

    esp_vfs_fat_sdmmc_mount_config_t mount_config = {};
    mount_config.format_if_mount_failed = false;
    mount_config.max_files = 5;
    mount_config.allocation_unit_size = 16 * 1024;

    esp_err_t ret = esp_vfs_fat_sdspi_mount(mount_point, &host, &slot_config, &mount_config, &card);

    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Falha ao montar SD Card (0x%x). Verifique os pinos/cartao.", ret);
        return ret;
    }

    sd_initialized = true;
    msc_active = false;
    ESP_LOGI(TAG, "SD Card montado com sucesso em %s", mount_point);
    return ESP_OK;
}

esp_err_t SdUsbManager::enable_usb_msc() {
    if (msc_active) return ESP_OK;

    // Trava anti-crash. Se nÃ£o houver SD Card, o botÃ£o nÃ£o faz nada e nÃ£o reinicia a placa.
    if (card == nullptr) {
        ESP_LOGE(TAG, "Cartao SD nao foi inicializado no boot! Abortando MSC para evitar Crash.");
        return ESP_FAIL;
    }

    ESP_LOGI(TAG, "Iniciando TinyUSB MSC...");
    
    tinyusb_config_t tusb_cfg = {};
    
    esp_err_t ret = tinyusb_driver_install(&tusb_cfg);
    if (ret != ESP_OK && ret != ESP_ERR_INVALID_STATE) {
        ESP_LOGE(TAG, "Falha ao instalar TinyUSB");
        return ret;
    }

    tinyusb_msc_sdmmc_config_t msc_config = {};
    msc_config.card = card;

    ret = tinyusb_msc_storage_init_sdmmc(&msc_config);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Falha ao vincular SD ao TinyUSB");
        return ret;
    }

    msc_active = true;
    ESP_LOGI(TAG, "Modo Pendrive Ativo! Conecte no PC.");
    return ESP_OK;
}

esp_err_t SdUsbManager::disable_usb_msc() {
    if (!msc_active) return ESP_OK;

    ESP_LOGI(TAG, "Saindo do Modo Pendrive. O sistema sera reiniciado para restaurar o Monitor Serial...");

    // 1. Derruba fisicamente os pinos USB internos do ESP32-S3 (GPIO 19 e 20)
    // Isso forÃ§a o Windows a registrar a remoÃ§Ã£o do Pendrive IMEDIATAMENTE.
    gpio_reset_pin(GPIO_NUM_19);
    gpio_reset_pin(GPIO_NUM_20);
    gpio_set_direction(GPIO_NUM_19, GPIO_MODE_INPUT);
    gpio_set_direction(GPIO_NUM_20, GPIO_MODE_INPUT);

    // 2. Aguarda 2.5 segundos para o PC tocar o som de "desconectado" e limpar a porta COM
    vTaskDelay(pdMS_TO_TICKS(2500));

    // 3. Reinicia o ESP32 graciosamente. O Bootloader vai restaurar o USB JTAG/Serial automaticamente.
    esp_restart();

    return ESP_OK; // O cÃ³digo nunca chegarÃ¡ aqui devido ao reboot
}
